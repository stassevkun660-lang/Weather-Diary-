import tkinter as tk
from tkinter import messagebox
import requests
import json
import os

class GitHubUserFinder:
    def __init__(self, root):
        self.root = root
        self.root.title("GitHub User Finder")
        
        # Список избранных пользователей
        self.favorite_users = []
        self.load_favorites()

        # Поле для ввода
        self.search_entry = tk.Entry(root, width=50)
        self.search_entry.pack(pady=10)

        # Кнопка поиска
        self.search_button = tk.Button(root, text="Поиск", command=self.search_user)
        self.search_button.pack()

        # Список результатов
        self.results_listbox = tk.Listbox(root, width=50)
        self.results_listbox.pack(pady=10)
        self.results_listbox.bind('<Double-1>', self.add_to_favorites)

        # Кнопка для отображения избранных пользователей
        self.show_favorites_button = tk.Button(root, text="Показать избранные", command=self.show_favorites)
        self.show_favorites_button.pack(pady=10)

    def search_user(self):
        username = self.search_entry.get().strip()
        if not username:
            messagebox.showerror("Ошибка", "Поле поиска не должно быть пустым.")
            return
        
        url = f"https://api.github.com/users/{username}"
        response = requests.get(url)

        if response.status_code == 200:
            user_data = response.json()
            self.results_listbox.delete(0, tk.END)
            self.results_listbox.insert(tk.END, f"{user_data['login']} (ID: {user_data['id']})")
        else:
            messagebox.showerror("Ошибка", "Пользователь не найден.")

    def add_to_favorites(self, event):
        selected = self.results_listbox.curselection()
        if selected:
            user_info = self.results_listbox.get(selected)
            if user_info not in self.favorite_users:
                self.favorite_users.append(user_info)
                self.save_favorites()
                messagebox.showinfo("Успех", "Пользователь добавлен в избранное.")
            else:
                messagebox.showinfo("Информация", "Пользователь уже в избранном.")

    def load_favorites(self):
        if os.path.exists("favorites.json"):
            with open("favorites.json", "r") as file:
                self.favorite_users = json.load(file)

    def save_favorites(self):
        with open("favorites.json", "w") as file:
            json.dump(self.favorite_users, file)

    def show_favorites(self):
        favorites_window = tk.Toplevel(self.root)
        favorites_window.title("Избранные пользователи")
        favorites_listbox = tk.Listbox(favorites_window, width=50)
        favorites_listbox.pack(pady=10)

        for user in self.favorite_users:
            favorites_listbox.insert(tk.END, user)


if __name__ == "__main__":
    root = tk.Tk()
    github_user_finder = GitHubUserFinder(root)
    root.mainloop()
