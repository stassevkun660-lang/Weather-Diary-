import tkinter as tk
from tkinter import ttk, messagebox
import random
import string
import json
import os

# Файл для хранения истории
HISTORY_FILE = "password_history.json"

def load_history():
    if os.path.exists(HISTORY_FILE):
        with open(HISTORY_FILE, "r") as f:
            return json.load(f)
    return []

def save_history(history):
    with open(HISTORY_FILE, "w") as f:
        json.dump(history, f)

def generate_password():
    length = int(length_slider.get())
    use_digits = digits_var.get()
    use_letters = letters_var.get()
    use_special = special_var.get()

    if not (use_digits or use_letters or use_special):
        messagebox.showwarning("Ошибка", "Выберите хотя бы один тип символов!")
        return

    chars = ""
    if use_letters:
        chars += string.ascii_letters
    if use_digits:
        chars += string.digits
    if use_special:
        chars += string.punctuation

    password = ''.join(random.choices(chars, k=length))
    password_entry.delete(0, tk.END)
    password_entry.insert(0, password)

    # Добавляем в историю
    history = load_history()
    history.append(password)
    save_history(history)
    update_history_table()

def update_history_table():
    for i in history_tree.get_children():
        history_tree.delete(i)
    for pwd in load_history():
        history_tree.insert("", "end", values=(pwd,))

# --- GUI ---
root = tk.Tk()
root.title("Random Password Generator")
root.geometry("500x400")

# Длина пароля
tk.Label(root, text="Длина пароля:").pack(pady=5)
length_slider = tk.Scale(root, from_=4, to=32, orient=tk.HORIZONTAL, length=300)
length_slider.set(12)
length_slider.pack()

# Чекбоксы
options_frame = tk.Frame(root)
options_frame.pack(pady=10)

digits_var = tk.BooleanVar(value=True)
letters_var = tk.BooleanVar(value=True)
special_var = tk.BooleanVar(value=True)

tk.Checkbutton(options_frame, text="Цифры", variable=digits_var).grid(row=0, column=0, padx=5)
tk.Checkbutton(options_frame, text="Буквы", variable=letters_var).grid(row=0, column=1, padx=5)
tk.Checkbutton(options_frame, text="Спецсимволы", variable=special_var).grid(row=0, column=2, padx=5)

# Кнопка генерации и поле вывода
tk.Button(root, text="Сгенерировать", command=generate_password).pack(pady=5)
password_entry = tk.Entry(root, width=40)
password_entry.pack(pady=5)

# История паролей
history_label = tk.Label(root, text="История паролей:")
history_label.pack()
history_tree = ttk.Treeview(root, columns=("password",), show="headings")
history_tree.heading("password", text="Пароль")
history_tree.column("password", width=450)
history_tree.pack(fill=tk.BOTH, expand=True)

update_history_table()

root.mainloop()
